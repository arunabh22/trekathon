import axios from "axios";

const GEMINI_API_URL = "https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent";
const API_KEY = process.env.REACT_APP_GEMINI_API_KEY;

const axiosInstance = axios.create({
  baseURL: GEMINI_API_URL,
  headers: {
    "Content-Type": "application/json",
  },
});

export const getAIResponse = async (userInput) => {
  try {
    const response = await axiosInstance.post(
      "",
      {
        contents: [
          {
            parts: [
              { text: userInput },
            ],
          },
        ],
      },
      {
        params: {
          key: API_KEY, // Pass API key as query parameter
        },
      }
    );

    // Check if response contains valid data
    if (!response.data?.candidates || response.data.candidates.length === 0) {
      throw new Error("No candidates returned in the response.");
    }

    // Extract AI response
    const aiResponse = response.data.candidates[0].outputText;
    return aiResponse;
  } catch (error) {
    console.error("Error fetching AI response:", error.message);
    return error.message.includes("candidates")
      ? "The AI was unable to generate a response."
      : "I'm sorry, I couldn't understand that.";
  }
};
